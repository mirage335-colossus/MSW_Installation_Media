What This Package Does
This package provides the device drivers in .inf form for ThinkPad computers, in order to allow you to
deploy Windows images with Microsoft System Center Configuration Manager (SCCM) by importing the device
drivers.  INF files will be available after executing the downloaded self-extracting EXE package.

var coll = document.getElementsByClassName("collapsible");
    var i;
    
    for (i = 0; i < coll.length; i++) {
      coll[i].addEventListener("click", function() {
        this.classList.toggle("active");
        var content = this.nextElementSibling;
        if (content.style.display === "block") {
          content.style.display = "none";
        } else {
          content.style.display = "block";
        }
      });
    }

    function showMoreModels() {
      document.getElementById('remainingItems').style.display = 'block';
      document.getElementById('viewMore').style.display = 'none';
      
    }
    function showLessModels() {
      document.getElementById('remainingItems').style.display = 'none';
      document.getElementById('viewMore').style.display = 'block';
